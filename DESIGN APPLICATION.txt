    <script>
        const ADMIN_PASS = "ARCHER2026"; 

        function verifyAccess() {
            const inputKey = document.getElementById('auth-passcode').value.trim();
            if (inputKey === ADMIN_PASS) {
                document.getElementById('auth-lock-overlay').classList.add('hidden');
                document.getElementById('studio-workspace').classList.remove('hidden');
            } else {
                document.getElementById('auth-error').classList.remove('hidden');
            }
        }

        function lockStudio() {
            document.getElementById('auth-lock-overlay').classList.remove('hidden');
            document.getElementById('studio-workspace').classList.add('hidden');
        }

        // Synchronize Input Fields to Canvas
        const sync = (inputId, displayId, prefix = '') => {
            const element = document.getElementById(inputId);
            if (element) {
                element.addEventListener('input', (e) => {
                    document.getElementById(displayId).innerText = prefix + e.target.value;
                });
            }
        };
        sync('input-headline', 'headline-wrapper');
        sync('input-address', 'display-address');
        sync('input-price', 'display-price', '💲 Price: ');
        sync('input-acres', 'display-acres', '📐 Size: ');
        sync('input-description', 'display-description');

        function parseMLSData() {
            const rawText = document.getElementById('mls-raw-input').value;
            if (!rawText.trim()) return;
            const lotMatch = rawText.match(/lot\s*#?\s*(\d+[A-Za-z]?)/i);
            const lotNum = lotMatch ? `Lot ${lotMatch[1]}` : 'Lot --';
            const cszMatch = rawText.match(/([A-Za-z\s]+),\s*([A-Z]{2})\s*(\d{5})/);
            let address = lotNum;
            if (cszMatch) address = `${lotNum}, ${cszMatch[1].trim()}, ${cszMatch[2].trim()} ${cszMatch[3].trim()}`;
            document.getElementById('input-address').value = address;
            document.getElementById('display-address').innerText = address;
            const priceMatch = rawText.match(/\$\s*[\d,]+/);
            if (priceMatch) {
                document.getElementById('input-price').value = priceMatch[0];
                document.getElementById('display-price').innerText = '💲 Price: ' + priceMatch[0];
            }
            generateAdCopy();
        }

        function generateAdCopy() {
            const headline = document.getElementById('input-headline').value;
            const address = document.getElementById('input-address').value;
            const price = document.getElementById('input-price').value;
            const acres = document.getElementById('input-acres').value;
            const desc = document.getElementById('input-description').value;

            const caption = `🏡 NEW LISTING!\n\n${headline}\n\n📍 Location: ${address}\n💰 Price: ${price}\n📐 Lot Size: ${acres}\n\n${desc}\n\n📩 Call/Text Patrick C. Washington at (502) 230-8636 for full details!`;
            document.getElementById('generated-ad-copy').value = caption;
        }

        function setFormat(format) {
            const template = document.getElementById('ad-template');
            template.classList.remove('format-feed', 'format-story', 'format-flyer');
            template.classList.add(`format-${format}`);
        }

        // PHOTO DISPLAY CONTROLLER
        function setMainAdImage(src) {
            const mainImg = document.getElementById('main-ad-photo');
            const placeholder = document.getElementById('photo-placeholder');
            const container = document.getElementById('image-container');
            
            mainImg.src = src;
            mainImg.classList.remove('hidden');
            if (placeholder) placeholder.classList.add('hidden');
            container.classList.remove('border-dashed');
            container.classList.add('border-solid');
        }

        function handleImageFiles(files) {
            if (!files || files.length === 0) return;
            
            const thumbContainer = document.getElementById('photo-thumbnails');
            thumbContainer.innerHTML = '';

            Array.from(files).forEach((file, index) => {
                if (!file.type.startsWith('image/')) return;

                const url = URL.createObjectURL(file);

                // Set the first image as the active ad display
                if (index === 0) {
                    setMainAdImage(url);
                }

                // Add thumbnail to thumbnail panel
                const thumb = document.createElement('img');
                thumb.src = url;
                thumb.className = 'w-12 h-12 object-cover rounded border-2 border-slate-700 hover:border-amber-400 cursor-pointer transition flex-shrink-0';
                thumb.onclick = () => setMainAdImage(url);
                thumbContainer.appendChild(thumb);
            });
        }

        // File Input Handler
        document.getElementById('image-upload').addEventListener('change', function(e) {
            handleImageFiles(e.target.files);
        });

        // Drag & Drop Handling on both Image Container and Panel
        const dropZones = [document.getElementById('image-container'), document.getElementById('panel-drop-zone')];

        dropZones.forEach(zone => {
            ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
                zone.addEventListener(eventName, (e) => {
                    e.preventDefault();
                    e.stopPropagation();
                }, false);
            });

            ['dragenter', 'dragover'].forEach(eventName => {
                zone.addEventListener(eventName, () => zone.classList.add('drop-active'), false);
            });

            ['dragleave', 'drop'].forEach(eventName => {
                zone.addEventListener(eventName, () => zone.classList.remove('drop-active'), false);
            });

            zone.addEventListener('drop', (e) => {
                const files = e.dataTransfer.files;
                handleImageFiles(files);
            }, false);
        });

        // Agent Headshot Upload
        document.getElementById('headshot-upload').addEventListener('change', function(e) {
            if (e.target.files[0]) {
                document.getElementById('ad-headshot').src = URL.createObjectURL(e.target.files[0]);
            }
        });

        // High-Res PNG Export
        function downloadAd() {
            const element = document.getElementById('ad-template');
            html2canvas(element, { 
                scale: 3, 
                useCORS: true, 
                allowTaint: true,
                logging: false 
            }).then(canvas => {
                const link = document.createElement('a');
                link.download = 'cwhomesky-graphic.png';
                link.href = canvas.toDataURL('image/png');
                link.click();
            });
        }
    </script>
