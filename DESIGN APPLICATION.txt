    <script>
        const ADMIN_PASS = "ARCHER2026"; 
        let activeSelectedLayer = null;

        function verifyAccess() {
            const inputKey = document.getElementById('auth-passcode').value.trim();
            if (inputKey === ADMIN_PASS) {
                document.getElementById('auth-lock-overlay').classList.add('hidden');
                document.getElementById('studio-workspace').classList.remove('hidden');
            } else {
                document.getElementById('auth-error').classList.remove('hidden');
            }
        }

        function lockStudio() {
            document.getElementById('auth-lock-overlay').classList.remove('hidden');
            document.getElementById('studio-workspace').classList.add('hidden');
        }

        // Live Input Sync
        const sync = (inputId, displayId, prefix = '') => {
            const el = document.getElementById(inputId);
            if (el) {
                el.addEventListener('input', (e) => {
                    document.getElementById(displayId).innerText = prefix + e.target.value;
                });
            }
        };
        sync('input-headline', 'headline-wrapper');
        sync('input-address', 'display-address');
        sync('input-price', 'display-price', '💲 Price: ');
        sync('input-acres', 'display-acres', '📐 Size: ');

        function parseMLSData() {
            const rawText = document.getElementById('mls-raw-input').value;
            if (!rawText.trim()) return;
            const lotMatch = rawText.match(/lot\s*#?\s*(\d+[A-Za-z]?)/i);
            const lotNum = lotMatch ? `Lot ${lotMatch[1]}` : 'Lot --';
            const cszMatch = rawText.match(/([A-Za-z\s]+),\s*([A-Z]{2})\s*(\d{5})/);
            let address = lotNum;
            if (cszMatch) address = `${lotNum}, ${cszMatch[1].trim()}, ${cszMatch[2].trim()} ${cszMatch[3].trim()}`;
            document.getElementById('input-address').value = address;
            document.getElementById('display-address').innerText = address;
            const priceMatch = rawText.match(/\$\s*[\d,]+/);
            if (priceMatch) {
                document.getElementById('input-price').value = priceMatch[0];
                document.getElementById('display-price').innerText = '💲 Price: ' + priceMatch[0];
            }
        }

        function setFormat(format) {
            const template = document.getElementById('ad-template');
            template.classList.remove('format-feed', 'format-story', 'format-flyer');
            template.classList.add(`format-${format}`);
        }

        // HANDLE FILES PROCESSOR
        function handleFiles(files) {
            if (!files || files.length === 0) return;
            Array.from(files).forEach((file) => {
                if (file.type.startsWith('image/')) {
                    const imgUrl = URL.createObjectURL(file);
                    createPhotoLayer(imgUrl);
                }
            });
        }

        // MANUAL FILE INPUT HANDLER
        document.getElementById('image-upload').addEventListener('change', function(e) {
            handleFiles(e.target.files);
        });

        // DRAG AND DROP EVENT ENGINE FOR CANVAS AND CONTROL PANEL
        const dropTargets = [document.getElementById('ad-template'), document.getElementById('control-drop-zone')];

        dropTargets.forEach(target => {
            ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
                target.addEventListener(eventName, preventDefaults, false);
            });

            ['dragenter', 'dragover'].forEach(eventName => {
                target.addEventListener(eventName, () => target.classList.add('drop-target-active'), false);
            });

            ['dragleave', 'drop'].forEach(eventName => {
                target.addEventListener(eventName, () => target.classList.remove('drop-target-active'), false);
            });

            target.addEventListener('drop', (e) => {
                const dt = e.dataTransfer;
                const files = dt.files;
                handleFiles(files);
            }, false);
        });

        function preventDefaults(e) {
            e.preventDefault();
            e.stopPropagation();
        }

        // CREATE MOVEABLE PHOTO LAYER
        function createPhotoLayer(src) {
            const wrapper = document.getElementById('photo-layers-wrapper');
            const imgContainer = document.createElement('div');
            imgContainer.className = 'draggable-layer layer-photo border border-transparent';
            imgContainer.style.left = '30px';
            imgContainer.style.top = '30px';
            imgContainer.style.zIndex = '1';

            const img = document.createElement('img');
            img.src = src;
            img.className = 'w-48 h-auto pointer-events-none rounded shadow-lg';
            imgContainer.appendChild(img);

            imgContainer.dataset.scale = "100";
            imgContainer.dataset.rotate = "0";

            imgContainer.addEventListener('click', (e) => {
                e.stopPropagation();
                selectPhotoLayer(imgContainer);
            });

            makeElementDraggable(imgContainer);
            wrapper.appendChild(imgContainer);
            selectPhotoLayer(imgContainer);
        }

        function selectPhotoLayer(layer) {
            document.querySelectorAll('.layer-photo').forEach(l => l.classList.remove('layer-selected'));
            activeSelectedLayer = layer;
            if (layer) {
                layer.classList.add('layer-selected');
                document.getElementById('photo-controls').classList.remove('hidden');
                document.getElementById('photo-scale').value = layer.dataset.scale || 100;
                document.getElementById('photo-rotate').value = layer.dataset.rotate || 0;
            } else {
                document.getElementById('photo-controls').classList.add('hidden');
            }
        }

        // PHOTO MODIFIER CONTROLS
        document.getElementById('photo-scale').addEventListener('input', (e) => {
            if (!activeSelectedLayer) return;
            activeSelectedLayer.dataset.scale = e.target.value;
            applyLayerTransforms(activeSelectedLayer);
        });

        document.getElementById('photo-rotate').addEventListener('input', (e) => {
            if (!activeSelectedLayer) return;
            activeSelectedLayer.dataset.rotate = e.target.value;
            applyLayerTransforms(activeSelectedLayer);
        });

        function applyLayerTransforms(layer) {
            const scale = (layer.dataset.scale || 100) / 100;
            const rotate = layer.dataset.rotate || 0;
            layer.style.transform = `scale(${scale}) rotate(${rotate}deg)`;
        }

        function moveLayerZ(direction) {
            if (!activeSelectedLayer) return;
            let currentZ = parseInt(activeSelectedLayer.style.zIndex || 1);
            activeSelectedLayer.style.zIndex = Math.max(1, currentZ + direction);
        }

        function deleteSelectedPhoto() {
            if (activeSelectedLayer) {
                activeSelectedLayer.remove();
                selectPhotoLayer(null);
            }
        }

        // DRAG & DROP POSITIONING ENGINE FOR CANVAS LAYERS
        function makeElementDraggable(el) {
            let pos1 = 0, pos2 = 0, pos3 = 0, pos4 = 0;
            el.onmousedown = dragMouseDown;

            function dragMouseDown(e) {
                e.preventDefault();
                pos3 = e.clientX;
                pos4 = e.clientY;
                document.onmouseup = closeDragElement;
                document.onmousemove = elementDrag;
            }

            function elementDrag(e) {
                e.preventDefault();
                pos1 = pos3 - e.clientX;
                pos2 = pos4 - e.clientY;
                pos3 = e.clientX;
                pos4 = e.clientY;
                el.style.top = (el.offsetTop - pos2) + "px";
                el.style.left = (el.offsetLeft - pos1) + "px";
            }

            function closeDragElement() {
                document.onmouseup = null;
                document.onmousemove = null;
            }
        }

        // INITIALIZE DRAGGABLE TEXT LAYERS
        document.querySelectorAll('.draggable-layer').forEach(el => makeElementDraggable(el));

        // HEADSHOT UPLOAD
        document.getElementById('headshot-upload').addEventListener('change', function(e) {
            if (e.target.files[0]) {
                document.getElementById('ad-headshot').src = URL.createObjectURL(e.target.files[0]);
            }
        });

        // EXPORT FUNCTION
        function downloadAd() {
            if (activeSelectedLayer) selectPhotoLayer(null);
            const element = document.getElementById('ad-template');
            html2canvas(element, { scale: 3, useCORS: true, allowTaint: true }).then(canvas => {
                const link = document.createElement('a');
                link.download = 'cwhomesky-graphic.png';
                link.href = canvas.toDataURL('image/png');
                link.click();
            });
        }
    </script>
